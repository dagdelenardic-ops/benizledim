Options -MultiViews -Indexes

# Set PHP version
<IfModule mime_module>
  AddHandler application/x-httpd-ea-php82 .php .php8 .phtml
  AddHandler application/x-httpd-alt-php82 .php .php8 .phtml
</IfModule>

<IfModule mod_rewrite.c>
  RewriteEngine On

  # Force HTTPS (proxy aware)
  RewriteCond %{HTTPS} !=on
  RewriteCond %{HTTP:X-Forwarded-Proto} !https [NC]
  RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

  # Never expose the internal public/ directory in the browser URL.
  # THE_REQUEST prevents this from firing on Laravel's internal rewrite.
  RewriteCond %{THE_REQUEST} \s/+public(?:[/?\s]) [NC]
  RewriteRule ^public(?:/)?(.*)$ /$1 [L,R=301,NE]

  # Normalize application routes before rewriting them into public/.
  # Without this, /admin/ enters public/.htaccess and becomes /public/admin.
  RewriteCond %{REQUEST_URI} !^/public/ [NC]
  RewriteCond %{DOCUMENT_ROOT}/public/$1 !-d
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule ^(.+)/$ /$1 [L,R=301,NE]

  # Skip rewrite when request already points to public.
  RewriteCond %{REQUEST_URI} ^/public/
  RewriteRule ^ - [L]

  # Rewrite everything to public/ subdirectory (transparent - no redirect)
  RewriteRule ^(.*)$ public/$1 [L]
</IfModule>
